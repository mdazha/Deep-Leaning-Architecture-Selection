\begin{frame}{Deep Learning for Time Series}

\small
\begin{columns}[T,totalwidth=\textwidth]
  %------------------ RNN ------------------
  \column{0.48\textwidth}
  \centering
  \includegraphics[width=0.8\linewidth]{Figures/architecture/rnn.png}
  \vspace{2pt}

  {
  \textbf{(a)RNN}\\
  \begin{itemize}
    \item Suffers from vanishing / exploding gradients.
    \item Hard to learn long-range dependencies.
    \item Lower accuracy than TCN and LSTM. 
  \end{itemize}
  }


  \column{0.48\textwidth}
  \centering

  \includegraphics[width=0.8\linewidth]{Figures/architecture/lstm.png}
  \vspace{2pt}

  {
  \textbf{(b) LSTM}\\
  \begin{itemize}
    \item Gating but adds many parameters and states.
    \item Still sequential in time: slow for long sequences.
    \item Can still forget very long-term context and uses large memory.
  \end{itemize}
  }
\end{columns}

\vspace{-0.1cm}
\textbf{Solution: Temporal Convolutional Networks (TCN)}
\footnote{Bai, Shaojie. "An Empirical Evaluation of Generic Convolutional and Recurrent Networks for Sequence Modeling." arXi}
\begin{itemize}
  \item  1D convolutions process all time
        steps in parallel, removing the sequential bottleneck of RNN/LSTM.
  \item The convolutions in the architecture are causal, meaning that there is         no information leakage from future to past
  \item Greatly reducing vanishing/exploding gradient issues with dilation,            capturing long-range dependencies more reliably 
  \item No recurrent gates; fewer parameters to store during training and simpler architecture to tune.
\end{itemize}

\end{frame}

\begin{frame}{TCN Basic Concepts}
%\footnotesize

\begin{columns}[T,onlytextwidth]
  \begin{column}{0.53\textwidth}
    \raggedright
    \begin{itemize}
        \item \textbf{Temporal Convolutional Networks (TCN)} are a deep learning architecture for processing sequential 1D data by using causal, dilated 1D convolutional layers % to capture patterns across time.

        \item \textbf{Causal convolution:} every timestep in the output can get information from the current/past timesteps% — it can't get information from future timesteps.

        \item \textbf{Receptive field} $RF$: set of entries of the input that affect a specific entry of the output.
        \[
            RF = 1 + l \cdot (k - 1)
        \]
    \end{itemize}
  \end{column}

\begin{equation}
RF = 1 + 2 \cdot \sum_{i=0}^{n-1} (k - 1) \cdot b^i = 1 + 2 \cdot (k - 1) \cdot \frac{b^n - 1}{b - 1}
\end{equation}

\begin{frame}{Bayesian TCN}
Let $\bs D$ be the training data,
\[
\bs D = \left\{ (G^{(k-l:k)},\, M^{(k-l:k)},\, I^{(k-l:k)},\, G^{(k+1:k+K_f)}) \right\}_{k=1}^{N}.
\]
Bayes' rule for the posterior:
\[
\pi_{\rm post}(\bs w \mid \bs D)
= \frac{\pi_{\rm like}(\bs D \mid \bs w)\,\pi_{\rm pr}(\bs w)}
       {\pi_{\rm evid}(\bs D)} .
\]
\begin{itemize}
    \item  Prior with  $\sigma_{\rm pr}^2$ variance,\\
\[
 \pi_{\rm pr}(\bs w) = \mathcal{N}\!\left(0,\, \sigma_{\rm pr}^2\right)
\]
    \item Likelihood under Gaussian output noise:
\[
\log \pi_{\rm like}(\bs D \mid \bs w)
=
-\frac{1}{2\sigma_{\rm noise}^{2}}
\sum_{t=1}^{N} 
\big\| G^{(k+1:k+K_f)} 
      - G_{\rm TCN}^{(k+1:k+K_f)}(\bs w)
\big\|_2^2
-\frac{N K_f}{2}\log(2\pi\sigma_{\rm noise}^{2})
\]

   \item The predictive distribution of the future blood glucose trajectory
$G^{\star (k+1:k+K_f)}$ is:
\begin{equation}
\pi\!\left(G^{\star (k+1:k+K_f)} \mid 
         G^{\star(k-l:k)}, M^{\star(k-l:k)}, I^{\star(k-l:k)},\, \bs D\right)\\
=
\int \pi\!\left(G^{\star (k+1:k+K_f)} 
           \mid G^{\star(k-l:k)}, M^{\star(k-l:k)}, I^{\star(k-l:k)},\, \bs w\right)
           \pi_{\text{post}}(\bs w \mid \bs D)\, d\bs w
\end{equation}

\end{itemize}
\end{frame}

\begin{frame}{Hierarchical Bayesian Inference}
\textbf{Bayesian TCN Model:}
\begin{itemize}
    \item Convolutional weight parameters: $\boldsymbol w $
    \item Inference hyperparameters \hspace{0.17 in}: $\boldsymbol \sigma = [\boldsymbol \sigma^{\text{pr}}, \boldsymbol \sigma^{\text{noise}}]$
    \item Architecture hyperparameters \hspace{0.02 in}: $\boldsymbol \xi \hspace{0.07 in} (\text{e.g. number of temporal blocks, dilation base, stride)} $
\end{itemize}
    \textbf{{Hierarchical Bayesian Inference:\footnotemark}}
\begin{itemize}
    \item \textit{Level 1.} Infer the network parameters $\boldsymbol w$,
    %
    \begin{equation*}\label{eq:lev1}
    \pi_{post} (\boldsymbol w | \boldsymbol D, \boldsymbol \sigma, \boldsymbol \xi) = 
    \frac{\pi_{like} (\boldsymbol D| \boldsymbol w, \boldsymbol \sigma, \boldsymbol \xi) \,
    \pi_{pr} (\boldsymbol w| \boldsymbol \sigma, \boldsymbol \xi)}
    {\pi_{evid} (\boldsymbol D| \boldsymbol \sigma, \boldsymbol \xi)}.
    \end{equation*}


    \item \textit{Level 2.} Infer the inference hyper-parameters $\boldsymbol \sigma$,
    %
    \begin{equation*}\label{eq:lev2}
    \pi_{post} (\boldsymbol \sigma | \boldsymbol D,  \boldsymbol \xi) = 
    \frac{\pi_{evid} (\boldsymbol D| \boldsymbol \sigma,  \boldsymbol \xi) \,
    \pi_{pr} (\boldsymbol \sigma |  \boldsymbol \xi)}
    {\pi_{evid} (\boldsymbol D|  \boldsymbol \xi)}.
    \end{equation*}


    \item \textit{Level 3.} Infer the archtecture hyper-parameters $\boldsymbol \xi$,
    %
    \begin{equation*}\label{eq:lev3}
    \pi_{post} (\boldsymbol \xi | \boldsymbol D) =
    \frac{\pi_{evid} (\boldsymbol D| \boldsymbol \xi) \,
    \pi_{pr} (\boldsymbol \xi)}{\pi_{evid} (\boldsymbol D)}.
    \end{equation*}

\end{itemize}
\footnotetext[7]{
Singh, Pratyush Kumar, Kathryn A. Farrell-Maupin, and Danial Faghihi. "A framework for strategic discovery of credible neural network surrogate models under uncertainty." Computer Methods in Applied Mechanics and Engineering 427 (2024): 117061.}
\end{frame}
%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%%
\begin{frame}{The Evidence Framework}
\begin{itemize}
\small
\item Variational Inference:
We approximate the posterior with a Gaussian Distribution
$q_{\bs\eta}(\bs w)$, where  $\bs\eta=(\mu_\theta,\sigma_\theta)$.
\[
\bs\eta^{\rm opt}
= \underset{\bs\eta}{\arg\min}\;
\mathcal{L}_{\rm VI}(\bs\eta)
\]
 Where, (for Gaussian-distributed total error)
    \begin{eqnarray}\label{eq:vi}
        \mathcal{L}_{\rm VI}(\bs \eta)
        & = & \mathrm{KL}\!\big[q_{\bs \eta}(\bs w) \,\|\, \pi_{\rm pr}(\bs w)\big]
        + \frac{Nf}{2} \log(2 \pi \sigma_{\rm noise}^2) \nonumber\\
        & + & \frac{1}{M_{MC}} \sum_{j=1}^{M_{MC}}
          \frac{1}{2\sigma_{\rm noise}^2}
          \big\| G^{(k+1:k+K_f)} - G^{(k+1:k+K_f)}_{\text{TCN}}(\hat{\bs w}^{(j)}) \big\|_2^2 
    \end{eqnarray}
\item The optimal values $\bs \sigma^{\rm opt}$ correspond to the Maximum A Posteriori (MAP) estimate:
\begin{equation}\label{eq:la_l2}
    \bs \sigma^{\rm opt} = \underset{\bs \sigma}{\rm argmax} \; 
    \mathcal{L}_{\rm Evid2}(\bs \sigma)
\end{equation}

The log-evidence for fixed $\bs \xi$ can be analytically evaluated as
\begin{equation}\label{eq:evid_l2}
\begin{aligned}
    \mathcal{L}_{\rm Evid2}(\bs \sigma) = \log \pi_{\rm evid}(\bs D \mid \bs \sigma, \bs \xi) 
    \;\approx\; 
    & -\frac{1}{2\sigma_{\rm noise}^2} 
      \sum_{t=1}^{N} 
      \big\| G^{(k+1:k+K_f)} - G^{(k+1:k+K_f)}_{\text{TCN}}(\bs \mu_{\rm post}) \big\|_2^2 \\
    & - \frac{Nf}{2}\log(2\pi \sigma_{\rm noise}^2) 
     - \frac{1}{2\sigma_{\rm pr}^2}\|\bs \mu_{\rm post}\|_2^2 \\
    & - \frac{p}{2} \log(2\pi \sigma_{\rm pr}^2) 
      - \frac{1}{2} \log \det (\bs H)
\end{aligned}
\end{equation}

\item The hyperparameters $\bs \sigma$ around their MAP values $\bs \sigma^{\rm opt}$,
    %
    \begin{equation}\label{eq:evid_l3}
	\rm{Evid_3} =  \pi_{\rm evid} (\bs D| \bs \xi) = 
    \pi_{\rm evid} (\bs D| \bs \sigma^{\rm opt},  \bs \xi) \,
    \pi_{\rm pr} (\bs \sigma^{\rm opt} | \bs \xi)
    \left[ 
    2\pi \, \sqrt{Var(\sigma_{\rm pr}^2) Var(\sigma_{\rm noise}^2)}
    \right]
    \end{equation}
    %
\end{itemize}
\end{frame}

\begin{frame}{Synthetic Training Data}

\textbf{Bergman Minimal Model}\footnote{Nandi, Souransu, and Tarunraj Singh. "Differential flatness based run-to-run control of blood glucose for people with type 1 diabetes."}
\[
\dot{G}(t) = -(X(t) + p_1) G(t) + p_1 G_b + \frac{\textcolor{blue}{R_{ag}}(t)}{V_g}
\]
\[
\dot{X}(t) = -p_2 X(t) + p_3 (I(t) - I_b)
\]
\[
\dot{I}(t) =
\begin{cases}
 -p_4 I(t) + \gamma (G(t)-h)(t-t_m), & G(t)\!\ge h,\; t\!\ge t_m\\[4pt]
 -p_4 I(t), & \text{otherwise}
\end{cases}
\]

\textbf{Gut Absorption Model (Dalla Man)}
\[
\textcolor{blue}{R_{ag}}(t) = f\, k_{\text{abs}}\, q_{\text{gut}}(t)
\]
\[
\dot{q}_{\text{gut}} = -k_{\text{abs}} q_{\text{gut}} + k_{\text{empt}} q_{\text{sto2}}
\]
\[
\dot{q}_{\text{sto1}} = -k_{21} q_{\text{sto1}} + \textcolor{red}{M} \delta(t-t_m),
\qquad
\dot{q}_{\text{sto2}} = -k_{\text{empt}} q_{\text{sto2}} + k_{21} q_{\text{sto1}}
\]


\textbf{Key states and parameters:}
\begin{itemize}
\item \(G(t)\): plasma glucose,\quad \(X(t)\): insulin action,\quad \(I(t)\): plasma insulin
\item \(p_1\): glucose effectiveness,\; \(p_2,p_3\): insulin action dynamics,\; \(p_4\): insulin decay
\item \(G_b,I_b\): basal glucose/insulin,\; \(h\): secretion threshold,\; \(\gamma\): secretion gain
\item \(R_{ag}(t)\): gut-driven glucose appearance,\; \(V_g\): glucose distribution volume
\end{itemize}


\end{frame}

\begin{frame}{Identifying Initial Pool of TCN Architectures}
\begin{enumerate}
    \item The receptive field has to be greater than or equal to one day,

    \begin{equation}
    1440 \le RF_n \le 2000
    \end{equation}

\item Condition for no holes: $k \ge b$.
\end{enumerate}
\textbf{Additional assumptions:}
\begin{itemize}
\item Number of filters in each block starts from 2 and can go up as much as 512.\\
\item Upper bound on kernel size 15. \\

\begin{frame}{Bayesian TCN Validation Metrics}
\textbf{B-TCN:} 1 hour prediction of blood glucose  
\begin{itemize}
    \item B-TCN mean prediction $\hat{G}^{\text{TCN}}$ and prediction uncertainty (variance): $\sigma^{\text{TCN}}$
    \item Blood glucose measurements: $G^{\text{data}}$
    \item Sensor noise $\sigma^{\text{data}}$: manufacturing-specific of CGM (e.g., Dexcom), FDA-approved code provides true vs measured blood glucose
\end{itemize}
\textbf{ Negative log-prediction density }
\[
\sigma^2 = (\sigma^{\text{TCN}})^2 + (\sigma^{\text{data}})^2
\]
\[
NLPD = 
\frac{1}{N}\sum_{i=1}^N 
\left[
\frac{\left(\hat{G}^{\text{TCN}}_i - \hat{G}^{\text{data}}_i\right)^2}{\sigma_i^2}
+ \ln(\sigma_i^2)
\right]
\leq Tol
\]
$\hat{G}^{\text{TCN}}_i$ represents the Bayesian TCN mean prediction of blood glucose at time $i$, while $\hat{G}^{\text{data}}_i$ denotes the corresponding CGM-measured glucose value. The predictive uncertainty consists of the model-derived standard deviation $\sigma^{\text{TCN}}_i$ and the CGM sensor noise $\sigma^{\text{data}}$. 

\end{frame}

\begin{frame}{Digital Twin (DT)}
\begin{figure}

  \caption*{}
\end{figure}
\vspace{-15pt}
The DT learns blood-glucose (BG) dynamics from individualized continuous glucose monitoring (CGM) time series data and computes insulin dosing schedules that are optimized under uncertainty. The computational stack comprises below modules:
\begin{enumerate}
    \item \textbf{Bayesian Filter:} dynamically update both weights $\bs w$ and inference hyperparameters $\bs \sigma$ as new data arrives sequentially. 
    
    \item \textbf{Risk-averse optimal control:} computation of insulin dosing/timing that minimize deviation of the predicted BG trajectory from a reference for type-1 diabetes (T1D), with mean-variance objective function.

\end{enumerate}

\end{frame}

\begin{frame}{Bayesian Filter}
In a patient-specific digital twin, sequential data $\bs D^{(1)}, \bs D^{(2)}, \ldots$ arrive at times $t^{(1)}, t^{(2)}, \ldots$, with insulin input $I^{(k)}$ influencing blood glucose (BG) evolution.

A latent state can be defined,
\[
\bs v = [\bs w, \bs \sigma]
\]
$\bs v$ latent state, including TCN weights $\bs w$ and inference hyperparameters $\bs \sigma$.

\vspace{6pt}
\textbf{Bayesian filter:} At step $k+1$, the posterior is:
\[
\pi_{\rm post}(\bs v^{(k+1)} \mid \bs D^{(1:k+1)}) = \frac{
\pi_{\rm like}(\bs D^{(k+1)} \mid \bs v^{(k+1)}) \cdot
\pi_{\rm prior}(\bs v^{(k+1)} \mid \bs D^{(1:k)})
}{
\pi_{\text{evid}}(\bs D^{(k+1)}|\bs D^{(1:k)})
}
\]

\vspace{4pt}
\textbf{Recursive update:}
\[
\pi_{\rm post}(\bs v^{(k)} \mid \bs D^{(1:k)}) \longrightarrow
\pi_{\rm post}(\bs v^{(k+1)} \mid \bs D^{(1:k+1)})
\]
\textbf{Assumptions:}
\item The latent state follows a first-order Markov process,
\[
\pi(\bs v^{(k+1)} \mid \bs v^{(1:k)}) 
  = \pi(\bs v^{(k+1)} \mid \bs v^{(k)}, I^{(k)}),
\]
\item Current measurements are conditionally independent of past measurements given the current state,
\[
\pi_{\rm like}(\bs D^{(k+1)} \mid \bs D^{(1:k)}, \bs v^{(k+1)})
  = \pi_{\rm like}(\bs D^{(k+1)} \mid \bs v^{(k+1)}),
\]
\end{frame}

\begin{frame}{Optimal Control Under Uncertainty}
\small

The Bayesian TCN induces a predictive distribution of future glucose
\[
  \bs G^{(k+1:k+K_f)} 
  \sim p\!\left(\bs G^{(k+1:k+K_f)} 
        \mid \bs D^{(1:k)}, \bs I^{(k)}, \bs v^{(k)} \right),
\]
from which we compute (e.g.\ by Monte Carlo over $\bs v$) for each step $j=1,\dots,K_f$
\[
  \mu_{G,k+j}(\bs I^{(k)}) = \mathbb{E}\!\left[G^{(k+j)}\right],
  \qquad
  \sigma_{G,k+j}^2(\bs I^{(k)}) = \mathrm{Var}\!\left[G^{(k+j)}\right].
\]

\medskip
\textbf{Chance-constrained insulin optimization problem at time $k$.}

Let $G_{\rm lb}$ and $G_{\rm ub}$ be hypo-/hyper-glycemic thresholds, 
and $G^{\rm targ}_{k+j}$ a target trajectory (e.g.\ euglycemic). 
We choose $\bs I^{(k)}$ by solving
%
\begin{equation}\label{eq:optimal_control}
    \begin{aligned}
        & \bs I_{\rm opt}^{(t+1:t+f)} 
        = \underset{\bs I}{\rm argmin}\;\;
        && \left\|\, \mathbb{E}\!\left[G^{t+1:t+f} - G_{\rm target}^{t+1:t+f}\right] 
        + \beta \,\mathbb{V}\!\left[G^{t+1:t+f} - G_{\rm target}^{t+1:t+f}\right] \right\|_2  \\
        & \text{subject to}
        && \mathcal{P}\!\left\{\, G^{t+1:t+f} \;\geq\; G_{lb} \,\right\} \;\geq\; 1-\epsilon_1,  \\
        &&& \mathcal{P}\!\left\{\, G^{t+1:t+f} \;\leq\; G_{ub} \,\right\} \;\geq\; 1-\epsilon_2, 
        \quad \text{for } (t_M^i + 120 \,\mathrm{min}) \leq t \leq t_M^{i+1}, \\
        &&& I_{BA}(t) + I_{BO}(t) \;\leq\; I_{\max}.
    \end{aligned}
\end{equation}
Here, $\bs I = [I_{BA}, I_{BO}]$ are the basal and bolus insulin inputs, 
$G^{t+1:t+f}$ is the predicted BG trajectory over the horizon $f$, 
and $G_{\rm target}^{t+1:t+f}$ is the nominal BG profile of a healthy individual. The operators $\mathbb{E}[\cdot]$ and $\mathbb{V}[\cdot]$ denote expectation and variance, respectively, with respect to the B-TCN predictive distribution.  
\end{frame}
